{/*
  Story UI V2 workspace — the DOCS-PAGE surface.

  On Storybook 9+ with the manager addon wired (.storybook/manager.ts imports
  StoryUI/manager), the workspace runs as a manager page at ?path=/workspace/,
  and that is the surface to use: the manager never remounts when a story
  enters the index, so a generation's live narration and steps stay on screen
  for the whole run. This docs page re-renders from scratch on every index
  change — which every generation causes — and recovers the thread from the
  manifest afterwards.

  It stays for two reasons: Storybook 8 has no `storybook/manager-api`, and a
  project can leave the addon un-wired. addon-docs always uses React, so this
  page works even when the project itself is Vue, Angular, Svelte or Web
  Components. When the addon IS present, a banner at the top says so and
  links to the page.
*/}

import { Meta } from '@storybook/addon-docs/blocks';
import { useEffect, useState } from 'react';
/*
  Imported from the PACKAGE, not from a copy in this project.

  The workspace builds its own UI with @radix-ui/themes. When its source was
  copied into the consumer's src/, that dependency became the consumer's
  problem — a Carbon or MUI team was told to install Radix to use a
  design-system tool, which is a fair thing to refuse. Importing from the
  package resolves it inside the package instead, where it is already a
  declared dependency.

  It also means the workspace updates when the package does. Copied source
  went stale the moment anything shipped, and the only fix was recopying it by
  hand.
*/
import { Workspace } from '@tpitre/story-ui/workspace';
import { resolveApiBase } from '@tpitre/story-ui/api-base';

<Meta title="Story UI/Workspace" />

export const TAB_FLAG_KEY = 'story-ui-tab';

export const Mount = () => {
  const [apiBase, setApiBase] = useState(null);
  const [tabRoute, setTabRoute] = useState(null);

  useEffect(() => {
    // Same resolution order this page has always used (explicit env, runtime
    // override, same-origin when deployed behind a proxy, then the local
    // default), now shared with the manager page so the two can never disagree.
    setApiBase(resolveApiBase());
    // The manager addon leaves this flag when it registers the page. The
    // preview shares sessionStorage with the manager (same origin, same tab).
    try { setTabRoute(sessionStorage.getItem(TAB_FLAG_KEY)); } catch { /* private mode */ }
  }, []);

  /*
    When the manager page exists, this sidebar entry IS a link to it. Landing
    here, reading a toast, and clicking through was a detour to the real
    workspace — and this page remounts on every generation, so it is the
    worse place to work. Same-origin, same tab: the manager URL can be set
    directly. The page still renders the workspace when there is no manager
    page (the addon un-wired), or if the navigation was refused.
  */
  useEffect(() => {
    if (!tabRoute) return;
    try {
      // The sidebar entry means "take me to the workspace", not "resume the
      // story I had open": clear the remembered story so the page opens on
      // Home. A reload mid-work still restores the story, because nothing
      // clears the key on that path.
      try { sessionStorage.removeItem('story-ui-v2-active'); } catch {}
      const target = `${window.location.origin}/?path=${tabRoute}`;
      if (window.top && window.top.location.href !== target) window.top.location.href = target;
    } catch { /* cross-origin host: fall through to the inline workspace */ }
  }, [tabRoute]);

  if (!apiBase) return null;

  const openTab = (e) => {
    e.preventDefault();
    try {
      window.top.location.href = `${window.location.origin}/?path=${tabRoute}`;
    } catch {
      window.open(`/?path=${tabRoute}`, '_blank');
    }
  };

  /*
    "Open in Storybook" opens the story in a new tab; the workspace handles
    that itself. Passing `onOpenStory` here would REPLACE that with whatever
    the handler does — the previous handler navigated the top window away
    from the workspace, losing the conversation in progress.

    `appearance` follows the host by default: Storybook's theme when it can
    be read from the page, the OS preference otherwise. Pin it to "light" or
    "dark" for a project that wants the workspace fixed.
  */
  if (tabRoute) {
    return (
      <p style={{ font: '14px system-ui, sans-serif', padding: 24 }}>
        Opening the Story UI workspace… If nothing happens,{' '}
        <a href={`/?path=${tabRoute}`} onClick={openTab}>open it here</a>.
      </p>
    );
  }

  return (
    <>
      {tabRoute && (
        <div
          role="note"
          style={{
            position: 'fixed', top: 8, left: '50%', transform: 'translateX(-50%)', zIndex: 20,
            padding: '6px 12px', borderRadius: 999, font: '12px system-ui, sans-serif',
            background: 'rgba(30, 30, 30, 0.92)', color: '#fff', boxShadow: '0 2px 10px rgba(0,0,0,0.25)',
            whiteSpace: 'nowrap',
          }}
        >
          Story UI now runs as a Storybook tab, which keeps a generation on screen for the whole run.{' '}
          <a href={`/?path=${tabRoute}`} onClick={openTab} style={{ color: '#7dd3fc', fontWeight: 600 }}>Open it there</a>
          {' '}— this page stays as the fallback.
        </div>
      )}
      <Workspace apiBase={apiBase} appearance="auto" />
    </>
  );
};

<Mount />
